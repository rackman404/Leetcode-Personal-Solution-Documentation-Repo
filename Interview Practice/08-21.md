class Solution:
    def isPowerOfTwo(self, n: int) -> bool:
        base = 2
        exp = 0
        result = 0
        if (n < 0):
            return False
        
        while (True): #do while loop (accounts for 0 case)
            result = base**exp
            exp += 1
            if (result == n):
                return True
            elif(result > n):
                break
                
        return False #clearly not a power of two
