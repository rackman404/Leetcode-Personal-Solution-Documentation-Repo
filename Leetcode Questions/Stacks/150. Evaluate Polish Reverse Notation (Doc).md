'''
assumptions: the token is valid (i.e we won't just end up with something like 2 numbers in the stack after all operators are ran through)

for x in token:
    if x is number:
        push to stack
    if x is operator:
        pop the two latest elements
        evaluate based on operator
        push result back onto stack
return the only remaining element in stack

'''

import math

class Solution:
    def evalRPN(self, tokens: List[str]) -> int:
        token_stack = []

        for x in tokens:
            #print(*token_stack, sep=' ')
            if (x != '+' and x != '-' and x != '*' and x != '/'):
                #print (int(x))
                token_stack.append(int(x))
            else:
                
                num1 = token_stack.pop()
                num2 = token_stack.pop()
                if (x == '+'):
                    num2 = num2 + num1
                elif (x == '-'):
                    num2 = num2 - num1
                elif (x == '*'):
                    num2 = num2 * num1
                elif (x == '/'):
                    num2 = num2 / num1 #truncated towards zero 
                    if (num2 > 0):
                        num2 = math.floor(num2)
                    else:
                        num2 = math.ceil(num2)
                token_stack.append(num2)

        return token_stack.pop()
