class Solution:
    def convert(self, s: str, numRows: int) -> str:
        array = [["" for i in range(len(s))] for j in range(numRows)]

        currentRow = 0
        currentCol = 0
        zigzag = False

        if (numRows == 1):#edge case
            return s

        #place first then consider position, too complicated otherwise
        for x in s:
            array[currentRow][currentCol] = x

            if (zigzag == False):  
                if (currentRow != numRows - 1):
                    currentRow += 1
                else:
                    currentRow -= 1
                    currentCol += 1
                    zigzag = True
                
            elif (zigzag == True):
                #print ("zigging: " + x + " " + str(currentRow))

                if (currentRow != 0): #continue zig behaviour
                    currentRow -= 1
                    currentCol += 1
                else: #go back down one now (but not next col)
                    currentRow += 1
                    zigzag = False


        ans = ""
        for row in array:
            #print (row)
            for x in row:
                ans += x
        #print (ans)
        return ans
